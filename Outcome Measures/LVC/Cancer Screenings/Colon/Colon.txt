/* ---- Open the saved dataset ---- */
libname outlib "P:\cahpr\jkiel1\Plans Project\Output";

data merged_county_final;
    set outlib.merged_county_final;
run;

/* ================= Step 1: raw colon cancer screening flag ================= */
data merged_county_final;
    set merged_county_final;

    array proc{5} $ hbrchcp1-hbrchcp5;

    colon_screen_raw = 0;

    do i = 1 to 5;

        /* ---- colon: endoscopic CPT codes require modifier "33" ---- */
        if proc{i} in ("45378","45379","45380","45381","45384","45385","45386",
                       "45388","45389","45390","45391","45392","45393","45398")
            and prev_proc_modifier = 1 then colon_screen_raw = 1;

        if proc{i} in ("G0104","G0105","G0121","G0106","G0120","G0122","G0327") 
            then colon_screen_raw = 1;

        /* ---- FOBT ---- */
        if proc{i} in ("81528","82270","82272","82274","G0464","G0107","G0327","G0328") 
            then colon_screen_raw = 1;

        /* ---- other colorectal screening codes ---- */
        if proc{i} in ("00812","74263","0464U","0537U") then colon_screen_raw = 1;

    end;

    drop i;
run;

/* ================= Step 2: earliest colon screening claim date, per beneficiary ================= */
proc sql;
    create table bene_first_colon_screen as
    select bene_id_18900, min(hbfrom) as first_colon_screen_date format=date9.
    from merged_county_final
    where colon_screen_raw = 1
    group by bene_id_18900;
quit;

/* ================= Step 3: check ALL claims before that date for C18 diagnosis ================= */
proc sql;
    create table claims_with_first_colon_screen as
    select a.*, b.first_colon_screen_date
    from merged_county_final as a
    inner join bene_first_colon_screen as b
    on a.bene_id_18900 = b.bene_id_18900;
quit;

data claims_before_colon_screen_flagged;
    set claims_with_first_colon_screen;

    array diag{5} $ hbdiag1-hbdiag5;

    colon_exclusion_dx = 0;

    if hbfrom < first_colon_screen_date then do;
        do i = 1 to 5;
            if substr(diag{i},1,3) = "C18" then colon_exclusion_dx = 1;
        end;
    end;

    drop i;
run;

proc sql;
    create table bene_colon_exclusion_before as
    select bene_id_18900, max(colon_exclusion_dx) as colon_exclusion_before
    from claims_before_colon_screen_flagged
    group by bene_id_18900;
quit;

/* ---- Recode: if C18 found before first colon screening, zero out ALL that beneficiary's screening flags ---- */
proc sql;
    create table merged_county_final as
    select a.*, coalesce(b.colon_exclusion_before, 0) as colon_exclusion_before
    from merged_county_final as a
    left join bene_colon_exclusion_before as b
    on a.bene_id_18900 = b.bene_id_18900;
quit;

data merged_county_final;
    set merged_county_final;
    if colon_exclusion_before = 1 then colon_screen_step2 = 0;
    else colon_screen_step2 = colon_screen_raw;
run;

/* ================= Step 4: unique list of beneficiaries still flagged ================= */
proc sql;
    create table bene_still_colon_screened as
    select distinct bene_id_18900
    from merged_county_final
    where colon_screen_step2 = 1;
quit;

/* ================= Step 5: match onto Hballenc_updated_2021, check ALL 2021 claims for C18 ================= */
data claims_2021_colon_filtered;
    if _n_ = 0 then set bene_still_colon_screened (keep=bene_id_18900);

    if _n_ = 1 then do;
        declare hash hb(dataset:"bene_still_colon_screened(keep=bene_id_18900)");
        hb.defineKey("bene_id_18900");
        hb.defineDone();
    end;

    set claims.Hballenc_updated_2021 (keep=bene_id_18900 hbdiag0-hbdiag4);

    if hb.check() ne 0 then delete;
run;

data claims_2021_colon_flagged;
    set claims_2021_colon_filtered;

    array diag21{5} $ hbdiag0-hbdiag4;

    colon_exclusion_2021 = 0;

    do i = 1 to 5;
        if substr(diag21{i},1,3) = "C18" then colon_exclusion_2021 = 1;
    end;

    drop i;
run;

proc sql;
    create table bene_colon_exclusion_2021 as
    select distinct bene_id_18900
    from claims_2021_colon_flagged
    where colon_exclusion_2021 = 1;
quit;

/* ================= Step 6: recode -- anyone on the 2021 exclusion list gets ALL colon screenings set to 0 ================= */
proc sql;
    create table merged_county_final as
    select a.*, 
           case when b.bene_id_18900 is not null then 1 else 0 end as colon_exclusion_2021
    from merged_county_final as a
    left join bene_colon_exclusion_2021 as b
    on a.bene_id_18900 = b.bene_id_18900;
quit;

data merged_county_final;
    set merged_county_final;
    if colon_exclusion_2021 = 1 then colon_screen_step3 = 0;
    else colon_screen_step3 = colon_screen_step2;
run;

/* ================= Step 7: final flag -- age >85 requirement applied last ================= */
data merged_county_final;
    set merged_county_final;
    if hkage > 85 then colon_lowvalue_screen = colon_screen_step3;
    else colon_lowvalue_screen = 0;
run;

/* ================= Beneficiary-level: anyone with at least one LVC colon screening ================= */
proc sql;
    create table bene_lowvalue_colon as
    select bene_id_18900, max(colon_lowvalue_screen) as bene_lowvalue_colon_screen
    from merged_county_final
    group by bene_id_18900;
quit;
